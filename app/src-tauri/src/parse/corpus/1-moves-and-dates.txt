# Natural-language commands the organizer should handle. One per line:
#   command => expected intent JSON (null fields left out), or ! to refuse.
# Dates: "before X" = earlier than X starts; "after X" = from the day after X ends;
# "since X" = from X's first day; "until X" = through X's last day.
# corpus_tests.rs reads every file in this folder, in file-name order.

# 1. Moves, copies, and renames in everyday words
move the report.pdf file to my Documents folder => {"intent":"move","src":{"kind":"item","name":"report.pdf"},"to":"Documents"}
please move all the PDFs in Downloads over to Documents => {"intent":"move","src":{"kind":"set","folder":"Downloads","filter":{"type":"pdfs"}},"to":"Documents"}
could you copy notes.txt over to my Desktop => {"intent":"copy","src":{"kind":"item","name":"notes.txt"},"to":"Desktop"}
put the screenshot from yesterday into Documents => {"intent":"move","src":{"kind":"set","filter":{"type":"screenshots","day":"yesterday"}},"to":"Documents"}
send the invoice to Archive => {"intent":"move","src":{"kind":"item","name":"invoice"},"to":"Archive"}
transfer all installers from Downloads to the Trash folder => {"intent":"trash","src":{"kind":"set","folder":"Downloads","filter":{"type":"installers"}}}

copy my notes to Backup => {"intent":"copy","src":{"kind":"item","name":"notes"},"to":"Backup"}

move Downloads/invoice.pdf to Documents => {"intent":"move","src":{"kind":"item","name":"Downloads/invoice.pdf"},"to":"Documents"}
